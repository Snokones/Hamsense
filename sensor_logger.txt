/*
  ESP32-C6 multi-sensor station - temperature (MCP9808) + humidity/pressure
  (BME280).  NO-GPS VARIANT.

  *** LITE BUILD -- see the "lite-ui" branch note below. ***

  This is the no-GPS build of the base station. The ATGM336H and TinyGPS++
  are gone entirely. NTP is the only clock source, and the firmware no longer
  does any sea-level-pressure correction.

  How this differs from main:
  - The page (web_pages.h) is the condensed, mobile-first LITE variant: no
    world map, no theme picker, no 7-day panel, no ZIP entry, no forecast,
    and ZERO external CDN dependencies -- it loads fully on a phone joined
    only to the station's own "PotaSense" AP with no internet.
  - No ACCUWEATHER_API_KEY requirement and no GET /config endpoint. Both
    exist on main purely to feed the forecast panel this page doesn't have.
  - The station's fixed latitude/longitude/elevation are three constants at
    the top of the script in web_pages.h, rather than being resolved from a
    ZIP the user types. Set them once for your site. That is what drives
    sunrise/sunset and the pressure correction here.

  Consequences worth knowing:
  - Pressure sent over the WebSocket is RAW STATION PRESSURE, both for the
    outdoor BME280 ("p") and the indoor satellite ("in"."p"). The page
    applies the ICAO sea-level correction using its STATION.elevM constant.
    Do not re-add the correction here without also removing it from
    web_pages.h, or it will be applied twice.
  - NTP is the ONLY clock source. In "PotaSense" AP-fallback mode there is
    no internet, so the clock never syncs -- and because history writes are
    gated on a synced clock, no history is recorded while in AP mode. GPS
    used to cover that case. A DS3231 RTC is the cheap fix if it ever
    matters; re-adding GPS is not.
  - GET /history and the LittleFS ring are still here and still recording,
    even though the lite page draws no history panel. The JSON is available
    if you want it; keeping the firmware identical to main here avoids a
    third variant to maintain.

  Other features:
  - AsyncWebServer + WebSocket, single live page, HTTP Basic Auth
  - ESP-NOW receiver for the indoor satellite node (satellite_sensor.ino,
    a XIAO ESP32-C6 + BME280 that deep-sleeps and reports every minute);
    the IndoorReport struct here must match the satellite's exactly
  - 30s sampling, pushed live to connected clients over WebSocket
  - 7-day temperature history on flash (LittleFS), served at GET /history
  - Seven-segment display, WS auto-reconnect
  - WiFi: tries the home network (3 attempts), falls back to broadcasting
    its own AP ("PotaSense") with a DNS redirect if it can't connect

  Power: this board serves an always-on web page, so it never sleeps. The
  savings here come from deleting the GPS (~25-30mA) and running the CPU at
  80MHz. Deep/light sleep is deliberately NOT used -- it would kill both the
  web server and the USB CDC port used for firmware updates.

  Wiring (XIAO ESP32-C6):
    MCP9808:   I2C (SDA=D4, SCL=D5), 3V3/GND, default address 0x18
    BME280:    I2C (SDA=D4, SCL=D5), 3V3/GND, address 0x76 or 0x77
    (UART1 / D6 / D7 are now free -- nothing is connected to them.)

  Required libraries (Arduino Library Manager):
    "Adafruit BME280 Library" + dependency "Adafruit Unified Sensor"
    "Adafruit MCP9808 Library"
*/

// ------------------------------------------------------------
// Sensor fault reporting
// ------------------------------------------------------------
// Policy: log on STATE CHANGE, not on every cycle. The earlier version
// printed "not initialized" on every failed read, which at a 30s cadence
// buries every other message within minutes -- and it still never told you
// when a sensor came back. Each sensor now carries a health record, so the
// log reads as events (FAILED / RECOVERED) rather than a wall of identical
// lines, and a sensor that flaps is visible as flapping.
struct SensorHealth {
  const char*   name;
  uint8_t       addr;             // primary I2C address, probed when a fault is reported
  uint8_t       addrAlt;          // secondary address (BME280 strapping); 0 = none
  bool          ok;               // did the last read/init succeed?
  bool          everOk;           // has it worked at any point since boot?
  bool          failureAnnounced; // did we actually print a FAILED line for the current outage?
  uint32_t      consecFails;      // consecutive failures, reset on success
  uint32_t      totalFails;       // cumulative since boot
  unsigned long lastGoodMs;       // millis() of the last valid reading
  unsigned long lastReportMs;     // millis() of our last printed line for this sensor; 0 = never
};
SensorHealth mcpHealth = { "MCP9808", 0x18, 0x00, false, false, false, 0, 0, 0, 0 };
SensorHealth bmeHealth = { "BME280",  0x76, 0x77, false, false, false, 0, 0, 0, 0 };

// How often a still-broken sensor re-announces itself, counted in failed
// attempts. At a 30s read interval, 20 attempts is roughly every 10 minutes:
// often enough that a dead sensor is obvious in a log you scroll into hours
// later, rare enough that it doesn't drown the console.
const uint32_t FAULT_REPEAT_EVERY = 20;

// Minimum gap between printed lines for a single sensor. This exists for the
// marginal-connection case: a sensor that fails and recovers on alternating
// cycles would otherwise cost a multi-line FAILED block plus a RECOVERED line
// every cycle -- noisier than the naive per-cycle logging this replaced.
// The first-ever report for a sensor bypasses this, so a device that is
// missing at boot is announced immediately rather than ten minutes later.
const unsigned long FAULT_MIN_GAP_MS = 60000;

// Decode Wire.endTransmission()'s return code. This is the difference between
// "nothing is answering at that address" and "the bus itself is jammed" --
// two very different wiring faults that both surface as begin() returning
// false and nothing else.
const char* i2cErrorText(uint8_t code) {
  switch (code) {
    case 0: return "ok";
    case 1: return "data too long for buffer";
    case 2: return "NACK on address -- nothing is responding here";
    case 3: return "NACK on data -- device answered, then stopped";
    case 4: return "bus error -- check SDA/SCL wiring and pull-ups";
    case 5: return "timeout -- bus held low by a wedged device or a short";
    default: return "unrecognised error code";
  }
}

// Probe one address without transferring data. Returns the endTransmission code.
uint8_t i2cProbe(uint8_t addr) {
  Wire.beginTransmission(addr);
  return Wire.endTransmission();
}

// Full bus scan. Cheap (127 probes, a few ms) and it turns "sensor not found"
// into "here is exactly what IS on the bus", which separates a wrong address
// from a dead bus from a power fault at a glance.
void i2cScan() {
  Serial.println("[I2C] scanning bus...");
  int found = 0;
  for (uint8_t a = 1; a < 127; a++) {
    if (i2cProbe(a) == 0) {
      const char *known = "";
      if      (a == 0x18)               known = "   <- MCP9808 (temperature)";
      else if (a == 0x76 || a == 0x77)  known = "   <- BME280 (humidity/pressure)";
      Serial.printf("[I2C]   found 0x%02X%s\n", a, known);
      found++;
    }
  }
  if (found == 0) {
    Serial.println("[I2C]   NO DEVICES FOUND.");
    Serial.println("[I2C]   Check SDA=D4, SCL=D5, both sensors on 3V3 and GND,");
    Serial.println("[I2C]   and that at least one board provides bus pull-ups.");
  } else {
    Serial.printf("[I2C] scan complete -- %d device(s) present.\n", found);
  }
}

// Record the outcome of a read or init attempt and decide whether it is worth
// a line of Serial output. Call it on EVERY attempt, success or failure --
// it does the filtering.
void sensorReport(SensorHealth &h, bool nowOk, const char *detail) {
  unsigned long now = millis();

  if (nowOk) {
    if (!h.ok) {  // transition into working
      if (h.failureAnnounced) {
        Serial.printf("[SENSOR] %s RECOVERED after %lu consecutive failure(s).\n",
                      h.name, (unsigned long)h.consecFails);
        h.lastReportMs     = now;
        h.failureAnnounced = false;
      } else if (!h.everOk) {
        Serial.printf("[SENSOR] %s online.\n", h.name);
        h.lastReportMs = now;
      }
      // If the matching failure was rate-limited away, stay quiet about the
      // recovery too -- printing half of each flap is just noise.
    }
    h.ok          = true;
    h.everOk      = true;
    h.consecFails = 0;
    h.lastGoodMs  = now;
    return;
  }

  h.consecFails++;
  h.totalFails++;

  bool startOfOutage = (h.consecFails == 1);
  bool periodicDue   = (h.consecFails % FAULT_REPEAT_EVERY) == 0;
  bool everPrinted   = (h.lastReportMs != 0);
  bool tooSoon       = everPrinted && (now - h.lastReportMs) < FAULT_MIN_GAP_MS;

  // startOfOutage rather than "was previously ok": at boot h.ok is already
  // false, so keying off the transition alone would make a sensor that is
  // simply not plugged in fail completely silently.
  if ((startOfOutage || periodicDue) && !tooSoon) {
    Serial.printf("[SENSOR] %s FAILED (%lu consecutive, %lu total): %s\n",
                  h.name, (unsigned long)h.consecFails, (unsigned long)h.totalFails,
                  detail ? detail : "no detail given");

    // The multi-line diagnostic block is reserved for the two cases where it
    // earns its space: the first failure since boot (you are debugging wiring)
    // and the periodic re-announce during a sustained outage. Emitting it on
    // every transition made a sensor with a marginal connection noisier than
    // the naive per-cycle logging this replaced. It also means the I2C bus
    // isn't probed on every flap, which matters when the bus is the fault.
    if (h.totalFails == 1 || periodicDue) {
      if (h.everOk && h.lastGoodMs) {
        Serial.printf("[SENSOR]   last good reading was %lus ago.\n",
                      (unsigned long)((now - h.lastGoodMs) / 1000UL));
      }
      Serial.printf("[SENSOR]   I2C 0x%02X -> %s\n", h.addr, i2cErrorText(i2cProbe(h.addr)));
      if (h.addrAlt) {
        Serial.printf("[SENSOR]   I2C 0x%02X -> %s\n", h.addrAlt, i2cErrorText(i2cProbe(h.addrAlt)));
      }
    }
    h.lastReportMs     = now;
    h.failureAnnounced = true;
  }
  h.ok = false;
}

// ---- Intra-day temperature history (LittleFS, 7-day rolling) ----
// One MCP9808 sample every HISTORY_INTERVAL_MS, stored as {epoch, meanTempF}
// in a fixed-capacity ring file. To append, we overwrite the slot holding the
// OLDEST timestamp, so the file always holds the newest HISTORY_CAPACITY
// samples and needs no separate head pointer on flash to corrupt -- the
// timestamps are the only bookkeeping. Served as JSON at GET /history.
struct __attribute__((packed)) TempSample {
  uint32_t epoch;  // Unix seconds; only ever written with real (synced) time -- see loop() gating
  float    tF;     // mean outdoor temperature over the interval, deg F
};
// epoch 0 is the sentinel for an empty slot (reads back as the "oldest").
static_assert(sizeof(TempSample) == 8, "TempSample must stay 8 bytes -- /temps.bin layout depends on it");

const char* const   HISTORY_PATH        = "/temps.bin";
const unsigned long HISTORY_INTERVAL_MS = 900000UL; // 15 min between stored samples
const int           HISTORY_CAPACITY    = 700;      // ~7 days at 15 min (672) + margin
bool          littleFsOk    = false;
double        histSum       = 0.0; // running sum of this interval's readings
uint32_t      histCount     = 0;   // number of readings folded into histSum
unsigned long lastHistWrite = 0;   // millis() of the last flush to flash

// ---- Indoor satellite (ESP-NOW) ----
// Report packet -- MUST exactly match the struct in satellite_sensor.ino.
// Named struct (not an anonymous typedef) so it can be forward-declared at
// the top of the file for the IDE's auto-generated prototypes. Packed so
// both sides agree on the byte layout.
struct __attribute__((packed)) IndoorReport {
  float    tempF;
  float    humidity;
  float    pressureHPa;  // raw station pressure; forwarded raw, corrected in the browser
  uint32_t seq;
};
// Compile-time guarantee that the packed layout is exactly what the radio
// protocol expects (3 floats + uint32 = 16 bytes). If a future edit changes
// the struct on one side but not the other, this fails the build instead of
// silently producing garbled readings at runtime.
static_assert(sizeof(IndoorReport) == 16, "IndoorReport layout changed -- must match satellite_sensor.ino exactly");

// Indoor satellite state. Shared between the ESP-NOW receive callback
// (WiFi task) and loop()/pushSensorData() (main task). A 16-byte struct
// copy isn't atomic, so all access goes through the spinlock below --
// volatile alone can't prevent a reader landing mid-copy and getting
// half-old, half-new data.
portMUX_TYPE indoorMux = portMUX_INITIALIZER_UNLOCKED;
IndoorReport  indoorLast = {};
bool          indoorEverReceived = false;
unsigned long indoorLastMs = 0;
uint32_t      indoorLastPrintedSeq = 0; // loop()-side only, no lock needed
bool          indoorWasStale = false;   // loop()-side edge detector for link-lost logging

// Periodic one-line summary of every subsystem, so a console you scroll into
// hours later tells you the current state without needing to catch a
// transition. 10 minutes is frequent enough to be useful, rare enough to stay
// out of the way.
const unsigned long HEALTH_REPORT_MS = 600000;
unsigned long lastHealthReport = 0;

// If no report arrives for 3 minutes (three missed 1-minute cycles), the
// page shows the indoor panel as stale ("--") rather than silently
// displaying old data.
const unsigned long INDOOR_STALE_MS = 180000;

// ESP-NOW receive callback. Runs in the WiFi task context, so keep it
// minimal: copy the packet and timestamp it under the lock, nothing heavier.
// NOTE: this is the arduino-esp32 core 3.x signature (esp_now_recv_info_t);
// the older (const uint8_t *mac, ...) form from most tutorials won't
// compile on the 3.x core the C6 requires.
void onEspNowRecv(const esp_now_recv_info_t *info, const uint8_t *data, int len) {
  if (len != sizeof(IndoorReport)) return; // ignore anything that isn't ours
  portENTER_CRITICAL(&indoorMux);
  memcpy(&indoorLast, data, sizeof(IndoorReport));
  indoorLastMs       = millis();
  indoorEverReceived = true;
  portEXIT_CRITICAL(&indoorMux);
}

// Take a consistent snapshot of the indoor state for use outside the lock.
void snapshotIndoor(IndoorReport &out, bool &received, unsigned long &lastMs) {
  portENTER_CRITICAL(&indoorMux);
  memcpy(&out, &indoorLast, sizeof(IndoorReport));
  received = indoorEverReceived;
  lastMs   = indoorLastMs;
  portEXIT_CRITICAL(&indoorMux);
}

// ---- NTP / time sync ----
const char *NTP_SERVER_1 = "pool.ntp.org";
const char *NTP_SERVER_2 = "time.nist.gov";
// Pacific Northwest: America/Los_Angeles. POSIX TZ string handles PST/PDT DST automatically.
const char *TZ_STRING = "PST8PDT,M3.2.0,M11.1.0";

// Returns true and fills `outEpoch` with a valid Unix time if NTP has synced
// the clock. Falls back to millis()-since-boot and returns false so callers
// can flag data as unsynced. NTP is the only time source in this build, so
// this returns false for the whole session in AP-fallback mode.
bool getEpochSeconds(uint32_t &outEpoch) {
  time_t now;
  time(&now);
  // Anything before ~2024-01-01 means the RTC was never synced.
  if (now > 1704067200UL) {
    outEpoch = (uint32_t)now;
    return true;
  }
  outEpoch = millis() / 1000;
  return false;
}

// ------------------------------------------------------------
// Temperature history helpers (LittleFS /temps.bin ring)
// ------------------------------------------------------------

// Append one sample by overwriting the slot with the oldest timestamp (epoch
// 0 = empty = oldest). O(HISTORY_CAPACITY) scan, but it runs only once per
// 15-minute interval, and it means there is no head pointer on flash that
// could corrupt -- the file self-describes via its timestamps.
void historyAppend(uint32_t epoch, float tF) {
  if (!littleFsOk) return;
  File f = LittleFS.open(HISTORY_PATH, "r+");
  if (!f) {
    // Previously a silent return -- a full or failing flash would quietly
    // stop recording history with nothing in the log to explain the gap.
    Serial.printf("[HIST] ERROR: cannot open %s for writing. This sample is lost.\n", HISTORY_PATH);
    return;
  }
  uint32_t oldestEpoch = 0xFFFFFFFFUL;
  int      oldestSlot  = 0;
  for (int i = 0; i < HISTORY_CAPACITY; i++) {
    TempSample s;
    if (f.read((uint8_t*)&s, sizeof(s)) != (int)sizeof(s)) break;
    if (s.epoch < oldestEpoch) {
      oldestEpoch = s.epoch;
      oldestSlot  = i;
      if (s.epoch == 0) break; // an empty slot is as old as it gets -- take it
    }
  }
  TempSample rec = { epoch, tF };
  f.seek((uint32_t)oldestSlot * sizeof(TempSample), SeekSet);
  size_t wrote = f.write((const uint8_t*)&rec, sizeof(rec));
  f.close();
  if (wrote != sizeof(rec)) {
    Serial.printf("[HIST] ERROR: short write (%u of %u bytes) into slot %d -- flash full or failing.\n",
                  (unsigned)wrote, (unsigned)sizeof(rec), oldestSlot);
  }
}

// Serialize the stored samples to JSON: {"unit":"F","interval_s":900,
// "samples":[[epoch,tempF],...]} sorted ascending, filtered to the last 7
// days so downtime/clock gaps can't surface stale points. Called from the
// AsyncTCP task via the /history handler; that task serializes its callbacks,
// so the static scratch buffer below is not re-entered concurrently.
String historyToJson() {
  String out = "{\"unit\":\"F\",\"interval_s\":";
  out += String(HISTORY_INTERVAL_MS / 1000UL);
  out += ",\"samples\":[";
  if (littleFsOk) {
    uint32_t nowEpoch;
    bool synced = getEpochSeconds(nowEpoch);
    uint32_t cutoff = (synced && nowEpoch > 7UL * 86400UL) ? nowEpoch - 7UL * 86400UL : 0;

    // 700 * 8 = 5.6KB kept static (off the small AsyncTCP task stack).
    static TempSample buf[HISTORY_CAPACITY];
    int n = 0;
    File f = LittleFS.open(HISTORY_PATH, "r");
    if (f) {
      for (int i = 0; i < HISTORY_CAPACITY; i++) {
        TempSample s;
        if (f.read((uint8_t*)&s, sizeof(s)) != (int)sizeof(s)) break;
        if (s.epoch != 0 && s.epoch >= cutoff) buf[n++] = s;
      }
      f.close();
    }
    // Insertion sort by epoch -- n is small and the data is near-sorted.
    for (int i = 1; i < n; i++) {
      TempSample key = buf[i];
      int j = i - 1;
      while (j >= 0 && buf[j].epoch > key.epoch) { buf[j + 1] = buf[j]; j--; }
      buf[j + 1] = key;
    }
    for (int i = 0; i < n; i++) {
      if (i) out += ',';
      out += '[';
      out += String(buf[i].epoch);
      out += ',';
      out += String(buf[i].tF, 2);
      out += ']';
    }
  }
  out += "]}";
  return out;
}

// ------------------------------------------------------------
// Push live sensor data to all WebSocket clients.
// Includes the sync flag so the UI can warn if the clock hasn't synced yet.
// ------------------------------------------------------------
void pushSensorData() {
  uint32_t epoch;
  bool synced = getEpochSeconds(epoch);

  // 256 is comfortable for 5 top-level members + the 4-member "in" object.
  // (Was 448 in the GPS build, which also had to carry a 6-member "gps"
  // object; that object is gone, so this shrank with it.)
  StaticJsonDocument<256> doc;
  doc["h"]    = lastHumidity;
  doc["t"]    = lastTempF;
  doc["p"]    = lastPressureHPa; // RAW station pressure -- page applies SLP correction
  doc["ts"]   = epoch;
  doc["sync"] = synced ? 1 : 0;

  // Indoor satellite readings, if the node is alive (reported within the
  // last 3 minutes). Snapshot under the lock so a mid-copy WS push can't
  // read torn data.
  IndoorReport  indoorCopy;
  bool          indoorRcvd;
  unsigned long indoorMs;
  snapshotIndoor(indoorCopy, indoorRcvd, indoorMs);

  bool indoorOk = indoorRcvd && (millis() - indoorMs) < INDOOR_STALE_MS;
  JsonObject inObj = doc.createNestedObject("in");
  inObj["ok"] = indoorOk ? 1 : 0;
  if (indoorOk) {
    // Forwarded raw, exactly as the satellite measured it. The sea-level
    // correction that used to happen here needed an altitude, which the GPS
    // supplied; it now happens in the browser using the ZIP's elevation.
    // The page uses the OUTDOOR temperature for that correction (the air
    // column being modelled is outside), which it already has.
    inObj["t"] = indoorCopy.tempF;
    inObj["h"] = indoorCopy.humidity;
    inObj["p"] = indoorCopy.pressureHPa;
  }

  String json;
  serializeJson(doc, json);
  ws.textAll(json);
}

// ------------------------------------------------------------
// Setup
// ------------------------------------------------------------
void setup() {
  Serial.begin(115200);
  // Non-blocking USB CDC writes. In the GPS build this existed to stop CDC
  // backpressure from starving UART1; there's no UART1 any more, but keep it:
  // without it, Serial.printf() blocks for seconds when the USB port is
  // enumerated but nothing is reading it, which stalls loop() and the
  // WebSocket pushes with it.
  Serial.setTxTimeoutMs(0);
  delay(300); // keep this -- it's the window that makes a bad build interruptible

  // --- CPU frequency: power ---
  // 80MHz is the floor with WiFi active (the radio will not run below it) and
  // saves roughly 15-25mA versus the 160MHz default. Safe for USB Serial/JTAG
  // on the C6, so firmware updates over USB are unaffected. Do NOT go lower
  // while WiFi is up, and do NOT reach for deep/light sleep on this node --
  // it serves an always-on page and must keep the USB port enumerated.
  //
  // Set BEFORE any WiFi call, which is why it sits this early in setup().
  //
  // DEBUG HANDLE: if you ever see a crash inside the WiFi driver -- ppTask,
  // wifi_deinit, wDev_Rxbuf_Deinit, or a "CORRUPT HEAP" assert from
  // multi_heap_poisoning -- put this back to 160 as the FIRST thing you try.
  // 80MHz is a supported WiFi clock, but it halves the headroom of the WiFi
  // protocol task, and the heap corruption fixed in the WiFi block below
  // lived in precisely that task. Ruling the clock out takes one edit.
  const uint32_t CPU_FREQ_MHZ = 80;
  setCpuFrequencyMhz(CPU_FREQ_MHZ);
  Serial.printf("CPU frequency: %uMHz\n", getCpuFrequencyMhz());

  // --- Turn off the onboard yellow LED ---
  // XIAO ESP32-C6's built-in LED is on GPIO15 and is ACTIVE-LOW (the pin
  // sinks current for the LED), so HIGH = off, LOW = on -- the opposite of
  // most boards' LED_BUILTIN convention. Using the explicit pin number (15)
  // rather than the LED_BUILTIN macro, since at least one published XIAO
  // ESP32-C6 example used a hardcoded pin instead of LED_BUILTIN for this
  // board, suggesting the macro may not be reliably defined for this exact
  // variant across all core versions.
  // GPIO15 is also a strapping pin on the ESP32-C6, but that only matters
  // during the very start of boot (before user code runs), so setting it
  // here in setup() is safe.
  const int ONBOARD_LED_PIN = 15;
  pinMode(ONBOARD_LED_PIN, OUTPUT);
  digitalWrite(ONBOARD_LED_PIN, HIGH); // HIGH = LED off on this board

  Wire.begin();
  // The default I2C bus timeout on arduino-esp32 is already 50ms, which
  // bounds how long a wedged/clock-stretching sensor can hang loop().
  // Some core versions clamp Wire.setTimeOut() to a 1000ms floor regardless
  // of the value passed, so rather than rely on a specific low number here,
  // we just confirm the configured timeout in Serial output at boot --
  // check your console; if it reports much more than ~50-1000ms, consider
  // updating your arduino-esp32 core version.
  Serial.printf("I2C bus timeout: %ums\n", Wire.getTimeOut());

  // Enumerate the bus before touching either sensor. If a begin() fails
  // below, this scan is what tells you whether the device is absent, at an
  // unexpected address, or the whole bus is down.
  i2cScan();

  // --- LittleFS: persistent store for the temperature history (/temps.bin) ---
  // format-on-fail (true) so a blank or corrupt flash comes up usable instead
  // of leaving history permanently dead. If the history file is absent or the
  // wrong size (first boot, or HISTORY_CAPACITY changed), (re)create it
  // preallocated with zeroed records -- epoch 0 reads back as an empty slot.
  littleFsOk = LittleFS.begin(true);
  if (littleFsOk) {
    const size_t wantBytes = (size_t)HISTORY_CAPACITY * sizeof(TempSample);
    File f = LittleFS.open(HISTORY_PATH, "r");
    size_t haveBytes = f ? f.size() : 0;
    if (f) f.close();
    if (haveBytes != wantBytes) {
      File w = LittleFS.open(HISTORY_PATH, "w");
      if (w) {
        TempSample zero = { 0, 0.0f };
        for (int i = 0; i < HISTORY_CAPACITY; i++) w.write((const uint8_t*)&zero, sizeof(zero));
        w.close();
        Serial.printf("LittleFS: created %s (%u bytes, %d slots).\n",
                      HISTORY_PATH, (unsigned)wantBytes, HISTORY_CAPACITY);
      } else {
        Serial.println("ERROR: LittleFS could not create history file -- history disabled.");
        littleFsOk = false;
      }
    } else {
      Serial.printf("LittleFS mounted; %s present (%u bytes).\n", HISTORY_PATH, (unsigned)haveBytes);
    }
    lastHistWrite = millis(); // baseline so the first stored bucket spans a full interval
  } else {
    Serial.println("WARNING: LittleFS mount failed -- temperature history disabled.");
  }

  // BME280 breakouts ship with one of two I2C addresses depending on how
  // the SDO pin is strapped (0x76 if pulled low, which is the common
  // default; 0x77 if pulled high) -- try both rather than assuming.
  bmeOk = bme.begin(0x76);
  if (!bmeOk) {
    bmeOk = bme.begin(0x77);
  }
  if (bmeOk) {
    configureBme();
    sensorReport(bmeHealth, true, nullptr);
    Serial.println("[SENSOR]   humidity + pressure, forced/weather mode.");
  } else {
    sensorReport(bmeHealth, false, "begin() failed at both 0x76 and 0x77");
    Serial.println("[SENSOR]   will keep retrying every read cycle -- no reboot needed.");
  }

  // MCP9808 -- precision temperature sensor, I2C address 0x18 (default,
  // all address pins pulled low). If you've changed any address pins on
  // your breakout, update the address here accordingly.
  mcpOk = mcp.begin(0x18);
  if (mcpOk) {
    mcp.setResolution(3); // max resolution: 0.0625°C / read time ~250ms -- fine at a 30s interval
    sensorReport(mcpHealth, true, nullptr);
    Serial.println("[SENSOR]   temperature, resolution 3 (0.0625C).");
  } else {
    sensorReport(mcpHealth, false, "begin() failed at 0x18");
    Serial.println("[SENSOR]   if your breakout has address pins strapped, update the address here.");
  }

  // (No GPS UART in this build. D6/D7 and UART1 are unused and left
  // uninitialised -- nothing should be wired to them.)

  // --- WiFi: try the home network, fall back to our own AP if it's not available ---
  // 3 attempts (~15s each, ~45s worst case) before giving up on STA and
  // broadcasting our own network instead. If STA succeeds on any attempt,
  // AP/DNS never start at all -- behavior is identical to before.
  //
  // IMPORTANT -- do not "clean up" the disconnect calls below back into
  // WiFi.disconnect(true). The bool argument is `wifioff`: passing true stops
  // and tears the WiFi driver down, and doing that once per failed attempt
  // drove a heap corruption inside the driver's own deinit path:
  //
  //   ppTask -> ieee80211_ioctl_process -> wifi_deinit -> ic_deinit
  //          -> wDev_Rxbuf_Deinit -> esp_timer_delete -> multi_heap_free
  //          -> verify_allocated_region -> assert -> panic_abort
  //
  //   CORRUPT HEAP: Bad tail at 0x408317bc. Expected 0xbaad5678 got 0xc034e7ec
  //   assert failed: multi_heap_free multi_heap_poisoning.c:279 (head != NULL)
  //
  // It only reproduces on a boot where the home network is unreachable, since
  // that is the only path that repeatedly deinits the radio. A plain
  // disconnect just drops the association and leaves the driver initialised,
  // which is all the retry loop actually needs.
  WiFi.persistent(false); // don't rewrite credentials to NVS on every begin()
  WiFi.mode(WIFI_STA);
  const int WIFI_MAX_ATTEMPTS = 3;
  const uint32_t WIFI_ATTEMPT_TIMEOUT_MS = 15000;
  bool staConnected = false;

  for (int attempt = 1; attempt <= WIFI_MAX_ATTEMPTS && !staConnected; attempt++) {
    Serial.printf("Connecting to WiFi (attempt %d/%d)", attempt, WIFI_MAX_ATTEMPTS);
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    uint32_t attemptStart = millis();
    while (WiFi.status() != WL_CONNECTED && millis() - attemptStart < WIFI_ATTEMPT_TIMEOUT_MS) {
      delay(200);
      Serial.print(".");
    }
    staConnected = (WiFi.status() == WL_CONNECTED);
    if (!staConnected) {
      Serial.println(" failed.");
      WiFi.disconnect(false); // drop the association only -- see the note above
      delay(500);             // let the driver settle before re-associating
    }
  }

  if (staConnected) {
    Serial.println("\nConnected!");
    Serial.print("IP: ");
    Serial.println(WiFi.localIP());
    // First line of defense: let the WiFi stack retry on its own after
    // brief drops. The watchdog in loop() covers the cases this misses.
    WiFi.setAutoReconnect(true);
  } else {
    Serial.println("\nCouldn't join home WiFi after 3 attempts -- starting fallback AP instead.");
    // Again: no wifioff. WiFi.mode() performs the STA->AP switch itself, and
    // it does so without the full driver teardown that was corrupting the heap.
    WiFi.disconnect(false);
    delay(200);
    WiFi.mode(WIFI_AP);
    bool apOk = WiFi.softAP(FALLBACK_AP_SSID, FALLBACK_AP_PASSWORD);
    if (apOk) {
      runningAsAP = true;
      Serial.printf("Fallback AP \"%s\" started. Connect to it, then browse to: ", FALLBACK_AP_SSID);
      Serial.println(WiFi.softAPIP());
      // Answer every DNS lookup with our own IP, so any URL the phone tries
      // to load lands on this device instead of a DNS failure. This is the
      // simple version (no full captive-portal popup) -- the user opens
      // their browser and tries any address, and it redirects here.
      dnsServer.start(53, "*", WiFi.softAPIP());
    } else {
      Serial.println("ERROR: failed to start fallback AP. Device has no network connectivity.");
    }
  }

  // --- NTP time sync ---
  // Only meaningful with real internet access (STA mode) -- skip it
  // entirely in AP-only fallback mode rather than waiting on something
  // that can't succeed. getEpochSeconds() already handles "never synced"
  // gracefully (relative time + a sync=0 flag), so this isn't required for
  // the device to work, just for timestamps to be real-world-accurate.
  if (staConnected) {
    configTzTime(TZ_STRING, NTP_SERVER_1, NTP_SERVER_2);
    // mDNS: advertise as hamsense.local on the home network.
    // Users can browse to http://hamsense.local instead of the IP address.
    // Works natively on Mac/iPhone; Windows 10/11 with Chrome/Edge; most Android.
    if (MDNS.begin("hamsense")) {
      MDNS.addService("http", "tcp", 80);
      Serial.println("mDNS started -- device reachable at http://hamsense.local");
    } else {
      Serial.println("WARNING: mDNS failed to start.");
    }
  } else {
    Serial.println("Skipping NTP sync -- no internet access in fallback AP mode.");
  }

  // --- Web UI authentication (HTTP Basic Auth) ---
  // Gate EVERY web path -- the page, the /history JSON, and the WebSocket --
  // behind a username/password kept in secrets.h (WEB_USER / WEB_PASS). The
  // browser prompts once and caches the credentials for the origin, so the
  // same login covers the WS handshake automatically (the WS upgrade is an
  // HTTP GET to the same host). Gating the socket matters: without it, anyone
  // could skip the page and pull live readings straight from /ws.
  // NOTE: this is Basic Auth over plain HTTP -- credentials and data are only
  // base64-encoded, not encrypted. It keeps casual users off the page but is
  // sniffable by anyone on the same LAN; real confidentiality would need TLS,
  // which this async stack can't do practically. Pair with an isolated IoT
  // VLAN if the LAN isn't fully trusted.
  ws.setAuthentication(WEB_USER, WEB_PASS);
  ws.onEvent(onWsEvent);
  server.addHandler(&ws);

  server.on("/", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send_P(200, "text/html", index_html);
  }).setAuthentication(WEB_USER, WEB_PASS);

  // Temperature history as JSON for the page to plot. Behind the same Basic
  // Auth as "/" -- otherwise the readings would leak around the login.
  server.on("/history", HTTP_GET, [](AsyncWebServerRequest *request){
    request->send(200, "application/json", historyToJson());
  }).setAuthentication(WEB_USER, WEB_PASS);

  // (No /config endpoint on the lite build -- see the note near the top of
  // this file. It exists on main only to hand the AccuWeather key to the
  // forecast panel, which this page doesn't have. One less public surface.)

  // In fallback AP mode, redirect any unrecognized request to "/" -- this is
  // what makes "any URL the phone tries to load" land on the sensor page,
  // combined with the DNSServer answering every hostname with our own IP.
  // Only enabled in AP mode; in normal STA mode, unknown paths still get a
  // normal 404, so a typo'd URL or a missing endpoint is still visible as
  // an error rather than silently redirecting.
  if (runningAsAP) {
    server.onNotFound([](AsyncWebServerRequest *request) {
      request->redirect("/");
    });
  }

  server.begin();

  // --- ESP-NOW receiver for the indoor satellite ---
  // Works in both STA and AP-fallback modes; the satellite discovers our
  // channel by scanning for whichever SSID is on the air. Receiving needs
  // no peer registration -- just init + callback.
  if (esp_now_init() == ESP_OK) {
    esp_now_register_recv_cb(onEspNowRecv);
    Serial.println("ESP-NOW receiver ready (indoor satellite).");
    Serial.printf("Base STA MAC (put in satellite secrets): %s\n", WiFi.macAddress().c_str());
    if (runningAsAP) {
      // In AP fallback the receiving interface is the softAP, which has a
      // different MAC -- the satellite would need this one instead.
      Serial.printf("Base AP MAC (only if base stays in AP mode): %s\n", WiFi.softAPmacAddress().c_str());
    }
  } else {
    Serial.println("WARNING: esp_now_init failed -- indoor satellite disabled.");
  }
  // Note: the base's WiFi radio power-saves between router beacons, which
  // can make it miss a single ESP-NOW packet. The satellite compensates by
  // retrying up to 8 times per report. If indoor updates ever prove
  // unreliable, the heavier fix is WiFi.setSleep(false); here -- at the
  // cost of noticeably higher idle current on the base.
}

// ------------------------------------------------------------
// Loop
// ------------------------------------------------------------
void loop() {
  unsigned long now = millis();

  // DNS redirect only runs in fallback AP mode.
  if (runningAsAP) {
    dnsServer.processNextRequest();
  }

  // --- WiFi watchdog ---
  if (now - lastWifiCheck >= WIFI_CHECK_MS) {
    lastWifiCheck = now;
    if (!runningAsAP) {
      if (WiFi.status() == WL_CONNECTED) {
        wifiDownSince = 0; // link healthy
      } else {
        if (wifiDownSince == 0) {
          wifiDownSince = now;
          Serial.println("WiFi link down -- attempting reconnect...");
        }
        WiFi.reconnect(); // nudge the stack in case auto-reconnect stalled
        if (now - wifiDownSince >= WIFI_REBOOT_AFTER_MS) {
          // Down for 5+ minutes straight: restart and run the full boot
          // sequence. If the router is back we rejoin; if it's still gone
          // we land in AP fallback and stay locally usable either way.
          Serial.println("WiFi down 5+ minutes -- restarting device.");
          ESP.restart();
        }
      }
    } else {
      // Booted into AP fallback (router was down at boot). If nobody is
      // actively using the AP and we've been up a while, restart to retry
      // the home network -- otherwise a device that booted during a router
      // outage stays stranded in AP mode forever. Never yanks the AP out
      // from under a connected user.
      if (WiFi.softAPgetStationNum() == 0 && now >= AP_RETRY_STA_MS) {
        Serial.println("AP fallback idle -- restarting to retry home WiFi.");
        ESP.restart();
      }
    }
  }

  if (now - lastSensorRead >= SENSOR_INTERVAL_MS) {
    lastSensorRead = now;

    // Temperature from MCP9808 (primary, high-precision sensor).
    // A sensor that falls off the bus mid-session does NOT make begin() fail
    // again -- mcpOk stays true and readTempF() just starts returning NaN. So
    // validate every reading, not just the init. Without this, NaN flows into
    // the WebSocket JSON and the page's digit renderer (which does
    // Number(NaN).toFixed() then indexes a segment map) breaks on it.
    if (mcpOk) {
      float tF = mcp.readTempF();
      if (isnan(tF)) {
        sensorReport(mcpHealth, false, "read returned NaN -- sensor stopped responding");
        mcpOk = false;
      } else if (tF < -40.0f || tF > 257.0f) {
        // The MCP9808's own spec is -40..+125C (-40..257F). Outside that is a
        // fault word or a garbled transfer, not weather.
        char why[80];
        snprintf(why, sizeof(why), "reading %.1fF is outside the sensor's -40..257F range", tF);
        sensorReport(mcpHealth, false, why);
        mcpOk = false;
      } else {
        lastTempF = tF + TEMP_OFFSET_F;
        sensorReport(mcpHealth, true, nullptr);
      }
      // Note: on failure lastTempF keeps its last good value rather than being
      // zeroed. The [READ] line below prints "--" so the log doesn't imply the
      // stale number is live.
    } else {
      // Retry begin() so a sensor connected or reseated after boot recovers
      // without a reboot.
      mcpOk = mcp.begin(0x18);
      if (mcpOk) {
        mcp.setResolution(3); // match the boot path so a recovered sensor runs at full resolution
        sensorReport(mcpHealth, true, nullptr);
      } else {
        sensorReport(mcpHealth, false, "begin() retry failed -- device not responding");
      }
    }

    // --- Intra-day temperature history ---
    // Fold this 30s reading into the current interval's running mean, and
    // flush that mean to flash every HISTORY_INTERVAL_MS. Gated on a *synced*
    // clock: an unsynced timestamp (relative millis) would file the sample
    // under the wrong day, so we simply don't record until NTP has set real
    // time. Reuses the existing sensor cadence -- no extra timer.
    // NOTE: NTP is the only clock source in this build, so in AP-fallback
    // mode this gate never opens and no history is recorded. That is a known
    // and accepted consequence of dropping the GPS.
    if (littleFsOk && mcpOk) {
      uint32_t epochNow;
      if (getEpochSeconds(epochNow)) {
        histSum   += lastTempF;
        histCount += 1;
        if (millis() - lastHistWrite >= HISTORY_INTERVAL_MS && histCount > 0) {
          historyAppend(epochNow, (float)(histSum / histCount));
          histSum       = 0.0;
          histCount     = 0;
          lastHistWrite = millis();
        }
      }
    }

    // Humidity + pressure from BME280. Same silent-failure mode as the
    // MCP9808: the library keeps reporting success while the chip returns NaN.
    if (bmeOk) {
      // Forced mode: the sensor idles until we explicitly trigger one
      // conversion. Blocks ~10ms at 1x oversampling -- negligible at a
      // 30-second read interval.
      bme.takeForcedMeasurement();
      float hum  = bme.readHumidity();
      // RAW station pressure -- no sea-level reduction here. The ICAO
      // correction needs a station altitude, which used to come from the GPS
      // and now comes from the ZIP code's elevation, which only the browser
      // knows. web_pages.h applies:
      //   SLP = P_station * (1 + (0.0065 * alt_m) / T_K) ^ 5.2561
      // If you ever re-add the correction here, remove it there too.
      float hpa = bme.readPressure() / 100.0f;

      if (isnan(hum) || isnan(hpa)) {
        sensorReport(bmeHealth, false, "read returned NaN -- sensor stopped responding");
        bmeOk = false;
      } else if (hum < 0.0f || hum > 100.0f) {
        char why[80];
        snprintf(why, sizeof(why), "humidity %.1f%% is outside 0-100%%", hum);
        sensorReport(bmeHealth, false, why);
        bmeOk = false;
      } else if (hpa < 300.0f || hpa > 1100.0f) {
        // Deliberately wide: 300-1100 hPa spans a hurricane at sea level to
        // the summit of Everest, so this only trips on genuinely broken data.
        char why[88];
        snprintf(why, sizeof(why), "pressure %.1f hPa is outside the plausible 300-1100 hPa", hpa);
        sensorReport(bmeHealth, false, why);
        bmeOk = false;
      } else {
        lastHumidity    = hum;
        lastPressureHPa = hpa;
        sensorReport(bmeHealth, true, nullptr);
      }
    } else {
      bmeOk = bme.begin(0x76) || bme.begin(0x77);
      if (bmeOk) {
        configureBme(); // match the boot path on recovery
        sensorReport(bmeHealth, true, nullptr);
      } else {
        sensorReport(bmeHealth, false, "begin() retry failed at both 0x76 and 0x77");
      }
    }

    // One consolidated reading line. A sensor that is currently faulted prints
    // "--" rather than its last good value, so nothing in the log looks live
    // when it isn't.
    {
      char tBuf[16], hBuf[16], pBuf[20];
      if (mcpHealth.ok) snprintf(tBuf, sizeof(tBuf), "%.2fF", lastTempF);       else strcpy(tBuf, "--");
      if (bmeHealth.ok) snprintf(hBuf, sizeof(hBuf), "%.1f%%", lastHumidity);   else strcpy(hBuf, "--");
      if (bmeHealth.ok) snprintf(pBuf, sizeof(pBuf), "%.1f hPa", lastPressureHPa); else strcpy(pBuf, "--");
      Serial.printf("[READ] temp %s | humidity %s | station pressure %s\n", tBuf, hBuf, pBuf);
    }

    // Push to WebSocket clients as long as at least one sensor gave us data.
    // If both sensors failed we still push (zeros) so the page doesn't go
    // dark -- the Serial output above will make the failure visible on the bench.
    pushSensorData();
  }

  ws.cleanupClients();

  // Log indoor satellite reports here rather than in the ESP-NOW callback
  // (which runs in the WiFi task and should stay minimal).
  {
    IndoorReport  inCopy;
    bool          inRcvd;
    unsigned long inMs;
    snapshotIndoor(inCopy, inRcvd, inMs);

    if (inRcvd && inCopy.seq != indoorLastPrintedSeq) {
      // The satellite's sequence counter lives in RTC memory, which survives
      // deep sleep but not power loss -- so a counter that jumps BACKWARDS
      // means the node rebooted, and a forward jump means reports were lost
      // in the air. Both are worth knowing when the radio link is what you're
      // debugging; neither is visible from the web page.
      if (indoorLastPrintedSeq != 0) {
        if (inCopy.seq < indoorLastPrintedSeq) {
          Serial.printf("[SAT] sequence reset (%lu -> %lu) -- satellite lost power or was reflashed.\n",
                        (unsigned long)indoorLastPrintedSeq, (unsigned long)inCopy.seq);
        } else if (inCopy.seq > indoorLastPrintedSeq + 1) {
          Serial.printf("[SAT] %lu report(s) missed (seq %lu -> %lu).\n",
                        (unsigned long)(inCopy.seq - indoorLastPrintedSeq - 1),
                        (unsigned long)indoorLastPrintedSeq, (unsigned long)inCopy.seq);
        }
      }
      indoorLastPrintedSeq = inCopy.seq;
      if (indoorWasStale) {
        Serial.println("[SAT] link RECOVERED.");
        indoorWasStale = false;
      }
      Serial.printf("[SAT] %.1fF | %.1f%% | %.1f hPa | seq %lu\n",
                    inCopy.tempF, inCopy.humidity, inCopy.pressureHPa,
                    (unsigned long)inCopy.seq);
    }

    // Link-lost edge. The page already greys the indoor panel out after
    // INDOOR_STALE_MS; this puts the same transition on the console, which is
    // where you are looking when the satellite is the suspect.
    if (inRcvd && !indoorWasStale && (millis() - inMs) >= INDOOR_STALE_MS) {
      indoorWasStale = true;
      Serial.printf("[SAT] LINK LOST -- no report for %lus (expected one per minute).\n",
                    (unsigned long)((millis() - inMs) / 1000UL));
    }
  }

  // --- Periodic health summary ---
  if (now - lastHealthReport >= HEALTH_REPORT_MS) {
    lastHealthReport = now;
    IndoorReport  inCopy;
    bool          inRcvd;
    unsigned long inMs;
    snapshotIndoor(inCopy, inRcvd, inMs);
    Serial.printf("[HEALTH] up %lum | MCP9808 %s (%lu fails) | BME280 %s (%lu fails) | "
                  "satellite %s | LittleFS %s | WiFi %s | heap %u B\n",
                  (unsigned long)(now / 60000UL),
                  mcpHealth.ok ? "ok" : "FAULT", (unsigned long)mcpHealth.totalFails,
                  bmeHealth.ok ? "ok" : "FAULT", (unsigned long)bmeHealth.totalFails,
                  !inRcvd ? "never seen" : ((millis() - inMs) < INDOOR_STALE_MS ? "ok" : "STALE"),
                  littleFsOk ? "ok" : "FAULT",
                  runningAsAP ? "AP fallback" : (WiFi.status() == WL_CONNECTED ? "connected" : "DOWN"),
                  (unsigned)ESP.getFreeHeap());
  }

  // Yield instead of busy-spinning. The GPS build used 2ms because UART1 had
  // to be drained promptly; with no UART there is nothing time-critical left
  // in this loop -- AsyncWebServer and ESP-NOW are both interrupt/task
  // driven, not polled here -- so a longer nap lets the FreeRTOS idle task
  // run more and costs nothing. Don't push this much past ~10ms: the WiFi
  // watchdog and history flush are still evaluated from here.
  delay(10);
}
